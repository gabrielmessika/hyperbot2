def sha(path):
    with path.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()


def quantiles(values):
    values.sort()
    if not values:
        return None
    return {str(q): values[int(q * (len(values) - 1))] for q in (0.1, 0.5, 0.9, 0.99)}


def info(payload, name):
    start = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(['curl', '--fail-with-body', '--max-time', '25', '-sS',
                             'https://api.hyperliquid.xyz/info', '-H',
                             'Content-Type: application/json', '--data-binary',
                             json.dumps(payload)], capture_output=True, check=True)
    path = RAW / f'{name}.json'
    with path.open('xb') as f:
        f.write(result.stdout)
    requests.append({'request': payload, 'started_at': start,
                     'received_at': datetime.now(timezone.utc).isoformat(),
                     'raw_path': str(path), 'sha256': sha(path)})
    return json.loads(result.stdout)

def pnl_summary(values):
    wins = sum(x for x in values if x > 0)
    losses = -sum(x for x in values if x < 0)
    return {'count': len(values), 'net_pnl_usd': sum(values),
            'gross_wins_usd': wins, 'gross_losses_usd': losses,
            'profit_factor': wins / losses if losses else None}

book_path = Path('/workspaces/trident/server-data/hip4/logs/hip4_nautilus_shadow/book_snapshots.jsonl')
count = invalid = duplicate = out_of_order = pairs = 0
first = last = None
spreads = defaultdict(list)
latest = {}
last_pair = {}
pair_ask_sums = []
pair_bid_sums = []
latencies = []
markets = set()
h = hashlib.sha256()
with book_path.open('rb') as f:
    for line in f:
        h.update(line)
        row = json.loads(line)
        count += 1
        ts = row['ts_event']
        first = min(first, ts) if first else ts
        last = max(last, ts) if last else ts
        mid = row['market_id']
        key = (mid, row['side_name'])
        markets.add(mid)
        previous = latest.get(key)
        if previous and previous['ts_event'] == ts:
            duplicate += 1
            continue
        if previous and previous['ts_event'] > ts:
            out_of_order += 1
            continue
        bid, ask = row.get('best_bid'), row.get('best_ask')
        bid = float(bid) if bid not in (None, '') else None
        ask = float(ask) if ask not in (None, '') else None
        if bid is None or ask is None or not (0 < bid < ask < 1):
            invalid += 1
            latest.pop(key, None)
            continue
        row['best_bid'], row['best_ask'] = bid, ask
        latest[key] = row
        spreads[mid.split('_')[0]].append(ask - bid)
        if row.get('source_latency_ms') is not None:
            latencies.append(float(row['source_latency_ms']))
        other = latest.get((mid, 'NO' if row['side_name'] == 'YES' else 'YES'))
        if other and other['ts_event'] == ts and last_pair.get(mid) != ts:
            last_pair[mid] = ts
            pairs += 1
            pair_ask_sums.append(ask + other['best_ask'])
            pair_bid_sums.append(bid + other['best_bid'])
books = {'source': str(book_path), 'sha256': h.hexdigest(), 'tier': 'B',
         'rows': count, 'first_exchange_ts': first, 'last_exchange_ts': last,
         'distinct_markets': len(markets), 'duplicate_consecutive_exchange_ts': duplicate,
         'out_of_order': out_of_order, 'invalid_rows_after_timestamp_filter': invalid,
         'unique_valid_observation_spread_probability_points': {
             k: {'count': len(v), 'quantiles': quantiles(v)} for k, v in spreads.items()},
         'source_latency_ms_quantiles': quantiles(latencies),
         'same_exchange_ts_complement_pairs': pairs,
         'pair_ask_sum_below_one': sum(x < 1 - 1e-9 for x in pair_ask_sums),
         'pair_bid_sum_above_one': sum(x > 1 + 1e-9 for x in pair_bid_sums),
         'pair_ask_sum_quantiles': quantiles(pair_ask_sums),
         'pair_bid_sum_quantiles': quantiles(pair_bid_sums),
         'limits': ['event-weighted, not time-weighted', 'same exchange timestamp is not simultaneous receipt',
                    'no fills, queue position, fees, markout or attainable return inferred',
                    'duplicate count tests consecutive timestamps per market-side only']}
